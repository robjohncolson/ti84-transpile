4. **Cross-reference with 0x0241A3** — the orchestrator's 9 wrappers (command bytes 0x00/0x01/0x04/0x05/0x09/0x0C/0x0D/0x16/0x17) are called via this table. At which table indices do they appear?

```javascript
const rom = fs.readFileSync('TI-84_Plus_CE/ROM.rom');
// Dump JP table starting at 0x0217B0
for (let addr = 0x0217B0; addr < 0x021900; addr += 4) {
  if (rom[addr] === 0xC3) {
    const target = rom[addr+1] | (rom[addr+2] << 8) | (rom[addr+3] << 16);
    console.log(`[${((addr - 0x0217B0)/4).toString().padStart(2)}] 0x${addr.toString(16)} → JP 0x${target.toString(16).padStart(6,'0')}`);
  } else {
    console.log(`[${((addr - 0x0217B0)/4).toString().padStart(2)}] 0x${addr.toString(16)} → NOT JP: ${rom[addr].toString(16)}`);
    break;
  }
}
```
