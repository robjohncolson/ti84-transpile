## What We Know

- The value at D00842 is in RAM, so it's set during boot — we need to trace what writes there

// Load ROM

// Register transpiled blocks
for (const [addr, fn] of Object.entries(transpiled)) {
  cpu.transpiledBlocks[parseInt(addr)] = fn;
}

// Boot
cpu.pc = 0x000000;
let steps = 0;
while (steps < 50000) {
  cpu.step();
  steps++;
}

// Read D00842
const d00842 = [];
for (let i = 0; i < 9; i++) d00842.push(cpu.mem[0xD00842 + i]);
console.log('D00842 bytes:', d00842.map(b => b.toString(16).padStart(2, '0')).join(' '));
