```javascript
function nDeriv(expr, variable, value, h) {
  h = h || 0.001;
  const f = new Function(variable, 'return ' + expr);
  return (f(value + h) - f(value - h)) / (2 * h);
}

function fnInt(expr, variable, a, b, n) {
  n = n || 1000;
  const f = new Function(variable, 'return ' + expr);
  const dx = (b - a) / n;
  let sum = 0;
  for (let i = 0; i < n; i++) {
    sum += f(a + (i + 0.5) * dx);
  }
  return sum * dx;
}
```

**IMPORTANT**: The nDeriv/fnInt implementations above are approximate — they handle the most common use case (nDeriv(expression, X, value)). The TI-84's syntax is `nDeriv(expression, variable, value[, h])` and `fnInt(expression, variable, lower, upper)`. Match this calling convention.
