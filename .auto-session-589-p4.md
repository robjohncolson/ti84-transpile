## Scanning approach
The 3-byte little-endian encoding of D01D0B is: 0x0B 0x1D 0xD0. Scan the ROM for this byte pattern, then verify each hit is inside an instruction that actually references this address (not a coincidental byte match in immediate data).

```javascript
import { readFileSync } from 'fs';
import { decodeInstruction } from './ez80-decoder.js';

const rom = readFileSync('TI-84_Plus_CE/ROM.rom');
const TARGET = 0xD01D0B;
const targetBytes = [TARGET & 0xFF, (TARGET >> 8) & 0xFF, (TARGET >> 16) & 0xFF]; // [0x0B, 0x1D, 0xD0]

// Scan for the 3-byte pattern
const hits = [];
for (let i = 0; i < rom.length - 2; i++) {
  if (rom[i] === targetBytes[0] && rom[i+1] === targetBytes[1] && rom[i+2] === targetBytes[2]) {
    hits.push(i);
  }
}

// For each hit, find the instruction containing it by trying decode from a few bytes before
for (const hit of hits) {
  // Try decoding from up to 6 bytes before the hit to find the instruction start
  for (let tryStart = Math.max(0, hit - 6); tryStart <= hit; tryStart++) {
    const inst = decodeInstruction(rom, tryStart, 'adl');
    if (tryStart + inst.length > hit && tryStart + inst.length <= hit + 3) {
      // This instruction spans our target bytes
      const hex = Array.from(rom.slice(tryStart, tryStart + inst.length))
        .map(b => b.toString(16).padStart(2, '0')).join(' ');
      console.log(`0x${tryStart.toString(16).padStart(6, '0')}  ${hex.padEnd(20)}  ${inst.asm}`);
      // Classify: check inst.asm for the address pattern
      break;
    }
  }
}
```
