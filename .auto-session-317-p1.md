// Search for CALL 0x05F685 (CD 85 F6 05) and JP 0x05F685 (C3 85 F6 05)
for (let i = 0; i < rom.length; i++) {
  if ((rom[i] === 0xCD || rom[i] === 0xC3) && rom[i+1] === 0x85 && rom[i+2] === 0xF6 && rom[i+3] === 0x05) {
    console.log(`${rom[i] === 0xCD ? 'CALL' : 'JP'} 0x05F685 at 0x${i.toString(16).padStart(6,'0')}`);
  }
}
// Also search for pointer references (little-endian 85 F6 05)
for (let i = 0; i < rom.length; i++) {

    console.log(`PTR ref to 0x05F685 at 0x${i.toString(16).padStart(6,'0')}`);
  }
