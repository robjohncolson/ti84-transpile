## How to boot the runtime

```javascript
import { readFileSync } from 'fs';
const ROM = readFileSync('TI-84_Plus_CE/ROM.rom');
const { createCPU } = await import('./TI-84_Plus_CE/cpu-runtime.js');
const { createPeripheralBus } = await import('./TI-84_Plus_CE/peripherals.js');

const peri = createPeripheralBus({ timerInterrupt: false });
const cpu = createCPU(peri);

// Load ROM
for (let i = 0; i < ROM.length; i++) cpu.mem[i] = ROM[i];

// Minimal init
cpu.mbase = 0xD0;
cpu.sp = 0xD1A87E;
cpu.iy = 0xD00080;
```

## Static ROM scanning pattern

```javascript
// Example: scan for LD (0xD00826),A
// eZ80 ADL mode: 32 26 08 D0 (LD (addr24),A where addr=0xD00826)
const target = [0x32, 0x26, 0x08, 0xD0];
for (let i = 0; i < ROM.length - 3; i++) {
    if (ROM[i] === target[0] && ROM[i+1] === target[1] && 
        ROM[i+2] === target[2] && ROM[i+3] === target[3]) {
        console.log(`LD (0xD00826),A at 0x${i.toString(16).padStart(6,'0')}`);
    }
}
```
