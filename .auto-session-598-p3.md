## File template for the probe
```javascript
#!/usr/bin/env node
import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import { decodeInstruction } from './ez80-decoder.js';

const __dirname = path.dirname(fileURLToPath(import.meta.url));
const romBytes = fs.readFileSync(path.join(__dirname, 'ROM.rom'));

function decode(startAddr, length) {
  let offset = startAddr;
  const end = startAddr + length;
  const lines = [];
  while (offset < end) {
    const insn = decodeInstruction(romBytes, offset, true);
    const bytes = [];
    for (let i = 0; i < insn.length; i++) bytes.push(romBytes[offset + i].toString(16).padStart(2, '0'));
    const line = `0x${offset.toString(16).padStart(6, '0')}  ${bytes.join(' ').padEnd(20)}  ${insn.mnemonic} ${insn.operands || ''}`;
    lines.push(line);
    offset += insn.length;
    if (insn.mnemonic === 'RET' && !insn.operands) break; // unconditional RET = function end (but continue if we haven't decoded enough)
  }
  return lines;
}

// Decode 0x05877A (general key handler) through 0x058EDA+
console.log('=== 0x05877A — General Key Handler ===');
decode(0x05877A, 400).forEach(l => console.log(l));

console.log('\n=== 0x058EDA — Key Classifier ===');
decode(0x058EDA, 250).forEach(l => console.log(l));

console.log('\n=== 0x0587E9 — Convergence Point ===');
decode(0x0587E9, 150).forEach(l => console.log(l));
```
