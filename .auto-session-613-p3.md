## Context

```js
// temp-decode-08f5e1.mjs
import { decodeInstruction } from './TI-84_Plus_CE/ez80-decoder.js';
import fs from 'fs';
const rom = fs.readFileSync('TI-84_Plus_CE/ROM.rom');
let pc = 0x08F5E1;
const end = 0x08F6C0;
const lines = [];
while (pc < end) {
  const insn = decodeInstruction(rom, pc, true);
  const bytes = Array.from(rom.slice(pc, pc + insn.length)).map(b => b.toString(16).padStart(2,'0')).join(' ');
  lines.push(`0x${pc.toString(16).padStart(6,'0')}: ${bytes.padEnd(20)} ${insn.mnemonic}`);
  pc += insn.length;
  // Stop at unconditional RET
  if (insn.mnemonic === 'RET' || insn.mnemonic === 'RETI' || insn.mnemonic === 'RETN') break;
}
console.log(lines.join('\n'));
```
