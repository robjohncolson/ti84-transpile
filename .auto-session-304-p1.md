## Reference: How probes work in this project
```javascript
#!/usr/bin/env node
import { readFileSync } from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import { decodeInstruction } from './ez80-decoder.js';

const __dirname = path.dirname(fileURLToPath(import.meta.url));
const rom = readFileSync(path.join(__dirname, 'ROM.rom'));

function disasmRange(startAddr, maxInstructions) {
  let pc = startAddr;
  const results = [];
  for (let i = 0; i < maxInstructions; i++) {
    const inst = decodeInstruction(rom, pc);
    results.push({ addr: pc, ...inst });
    // Stop at RET, JP (not conditional), HALT
    if (inst.mnemonic === 'RET' && !inst.condCode) break;
    if (inst.mnemonic === 'JP' && !inst.condCode && inst.operands && !inst.operands.includes('(')) break;
    pc += inst.length;
  }
  return results;
}
```

The `decodeInstruction(buffer, offset)` function returns `{ mnemonic, operands, length }`. The ROM is loaded raw from ROM.rom. All addresses in the ROM are physical (file offset = address for first 4MB).
