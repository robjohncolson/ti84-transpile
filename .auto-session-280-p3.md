## How to run dynamic traces
Look at existing probes in the repo for the pattern. Key setup:
```javascript
import { readFileSync } from 'fs';
import { createCPU, createMemoryController } from './cpu-runtime.js';
import { createPeripheralBus } from './peripherals.js';

const rom = readFileSync('TI-84_Plus_CE/ROM.rom');
const mem = createMemoryController(rom);
const periph = createPeripheralBus({ timerInterrupt: false });
const cpu = createCPU(mem, periph);

// Set registers
cpu.registers.PC = 0x09EDC3;
cpu.registers.IY = 0xD1A860;
cpu.registers.IX = 0xD1A860;
cpu.registers.SP = 0xD1987F;
// ... seed RAM as needed ...

// Step and log
for (let i = 0; i < 500; i++) {
  const pc = cpu.registers.PC;
  cpu.step();
  // log pc, new PC, etc.
}
```
