## Key code patterns (from probe-phase359-ram-trampoline.mjs)

RAM trampoline injection:
```js
const RAM_THRESHOLD = 0xD00000;
function injectRamTrampoline(executor, targetPc) {
  const key = `${(targetPc >>> 0).toString(16).padStart(6, '0')}:adl`;
  if (executor.compiledBlocks[key]) return;
  executor.compiledBlocks[key] = (cpu) => {
    const mem = cpu.mem;
    const sp = cpu.sp & 0xFFFFFF;
    const retAddr = mem[sp] | (mem[sp + 1] << 8) | (mem[sp + 2] << 16);
    cpu.sp = (sp + 3) & 0xFFFFFF;
    return retAddr;
  };
}
```

Boot setup:
```js
const peripherals = createPeripheralBus({ timerInterrupt: false });
const executor = createExecutor(PRELIFTED_BLOCKS, mem, { peripherals });
// Pre-inject known RAM target
injectRamTrampoline(executor, 0xD18C22);
// Also inject dynamically in onMissingBlock for any new RAM targets
```
