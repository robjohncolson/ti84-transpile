# Task: Sweep B values 0x00-0xFF to find which makes 0x080151 return Z

## Context
Function 0x062055 is the OS entry to cxMain (home screen). At entry:
- `LD A,B` (A = B register)
- `CALL 0x080151` (entry gate)
- `JR NZ,0x0620CC` (if NZ → alternate path, crash)
- If Z → continues to cxMain dispatch

Session 260 found B=0 gives NZ (fails). We need to find which B value(s) give Z.

Additionally, the gate function might read RAM state (like D0230F or D007E0) that affects the result. After the B sweep, also try:
- B=0 with D0230F set to different values (0x3E, 0x3F, 0x40, 0x5E)
- B=0 with D007E0 set to 0x40 (cxMain mode)

## Deliverable
Create probe file `TI-84_Plus_CE/probe-phase261-080151-b-sweep.mjs` that:

1. Calls 0x080151 with A=B for each B in 0x00-0xFF
2. Records Z/NZ flag for each
3. Reports ALL values that return Z
4. Then tests B=0 with D0230F and D007E0 variants
5. For any Z-returning scenario, trace the full block path

## Technical details
- ROM file: `TI-84_Plus_CE/ROM.rom`
- Transpiled JS: `TI-84_Plus_CE/ROM.transpiled.js` (gunzip from .gz if needed)
- Runtime: `TI-84_Plus_CE/cpu-runtime.js`
- Use `createCPU` and `createPeripheralBus({ timerInterrupt: false })`
- D0230F = 0xD0230F, D007E0 = 0xD007E0 (RAM addresses)
- IY = 0xD00080 (standard OS IY base)
- MBASE = 0xD0 (default in cpu-runtime.js)

## Template
```javascript
import { readFileSync } from 'fs';
import { createCPU, createPeripheralBus } from './cpu-runtime.js';

const rom = readFileSync(new URL('./ROM.rom', import.meta.url));

function runGate(cpu, bus, aVal, label, ramSetup) {
  // Reset CPU
  cpu.pc = 0x080151;
  cpu.sp = 0xD1A860;
  cpu.a = aVal;
  cpu.b = aVal;
  cpu.iy = 0xD00080;
  cpu.madl = 1;
  
  if (ramSetup) ramSetup(cpu);
  
  // Push return sentinel
  const sentinel = 0x7FFFFE;
  cpu.sp -= 4;
  cpu.mem[cpu.sp] = sentinel & 0xFF;
  cpu.mem[cpu.sp+1] = (sentinel >> 8) & 0xFF;
  cpu.mem[cpu.sp+2] = (sentinel >> 16) & 0xFF;
  cpu.mem[cpu.sp+3] = cpu.madl ? 0x80 : 0x00;
  
  const blocks = [];
  let steps = 0;
  while (steps < 200 && cpu.pc !== sentinel) {
    blocks.push(cpu.pc);
    const block = cpu.blocks[cpu.pc];
    if (!block) { blocks.push('MISSING:' + cpu.pc.toString(16)); break; }
    block(cpu, cpu.mem, bus);
    steps++;
  }
  
  return { z: cpu.flags.z, steps, blocks, aOut: cpu.a };
}

const bus = createPeripheralBus({ timerInterrupt: false });
const cpu = createCPU(bus);
for (let i = 0; i < rom.length; i++) cpu.mem[i] = rom[i];
const transpiled = readFileSync(new URL('./ROM.transpiled.js', import.meta.url), 'utf-8');
new Function('cpu', 'mem', 'bus', transpiled)(cpu, cpu.mem, bus);

// Phase 1: Sweep all B values
console.log('=== B VALUE SWEEP (0x00-0xFF) ===');
const zValues = [];
for (let b = 0; b <= 0xFF; b++) {
  const r = runGate(cpu, bus, b, `B=0x${b.toString(16).padStart(2,'0')}`);
  if (r.z) {
    zValues.push(b);
    console.log(`B=0x${b.toString(16).padStart(2,'0')}: Z! steps=${r.steps}, A_out=0x${r.aOut.toString(16)}, blocks=${r.blocks.map(x=>typeof x==='number'?x.toString(16):x).join(',')}`);
  }
}
console.log(`\nZ-returning B values: ${zValues.length > 0 ? zValues.map(v=>'0x'+v.toString(16).padStart(2,'0')).join(', ') : 'NONE'}`);

// Phase 2: RAM state variants with B=0
console.log('\n=== RAM STATE VARIANTS (B=0) ===');
for (const [label, setup] of [
  ['D0230F=0x3E', c => { c.mem[0xD0230F] = 0x3E; }],
  ['D0230F=0x3F', c => { c.mem[0xD0230F] = 0x3F; }],
  ['D0230F=0x40', c => { c.mem[0xD0230F] = 0x40; }],
  ['D0230F=0x5E', c => { c.mem[0xD0230F] = 0x5E; }],
  ['D007E0=0x40', c => { c.mem[0xD007E0] = 0x40; }],
  ['D007E0=0x40+D0230F=0x3F', c => { c.mem[0xD007E0] = 0x40; c.mem[0xD0230F] = 0x3F; }],
]) {
  const r = runGate(cpu, bus, 0, label, setup);
  console.log(`${label}: ${r.z ? 'Z' : 'NZ'}, steps=${r.steps}, A_out=0x${r.aOut.toString(16)}, blocks=${r.blocks.map(x=>typeof x==='number'?x.toString(16):x).join(',')}`);
  // Reset RAM
  cpu.mem[0xD0230F] = 0;
  cpu.mem[0xD007E0] = 0;
}
```
