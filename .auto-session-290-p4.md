## How to decode instructions
```javascript
import { decodeInstruction } from './ez80-decoder.js';
import { readFileSync } from 'fs';

const rom = readFileSync('TI-84_Plus_CE/ROM.rom');
function disasmAt(startAddr, count) {
  let pc = startAddr;
  for (let i = 0; i < count; i++) {
    const instr = decodeInstruction(rom, pc);
    console.log(`0x${pc.toString(16).padStart(6,'0')}: ${instr.asm}  [${instr.bytes.map(b=>'0x'+b.toString(16).padStart(2,'0')).join(' ')}]`);
    pc += instr.length;
  }
  return pc;
}
```
