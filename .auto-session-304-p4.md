## Reference: How probes work
```javascript
#!/usr/bin/env node
import { readFileSync } from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';
import { decodeInstruction } from './ez80-decoder.js';

const __dirname = path.dirname(fileURLToPath(import.meta.url));
const rom = readFileSync(path.join(__dirname, 'ROM.rom'));

function disasmRange(startAddr, maxInstructions) {
  let pc = startAddr;
  const results = [];
  for (let i = 0; i < maxInstructions; i++) {
    const inst = decodeInstruction(rom, pc);
    results.push({ addr: pc, ...inst });
    if (inst.mnemonic === 'RET' && !inst.condCode) break;
    if (inst.mnemonic === 'JP' && !inst.condCode && inst.operands && !inst.operands.includes('(')) break;
    pc += inst.length;
  }
  return results;
}
```

`decodeInstruction(buffer, offset)` returns `{ mnemonic, operands, length }`. ROM loaded raw, file offset = address.
