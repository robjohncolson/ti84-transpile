```javascript
import fs from 'node:fs';
const rom = fs.readFileSync('TI-84_Plus_CE/ROM.rom');

// Dump function bytes at 0x03FC1C
const funcStart = 0x03FC1C;
const bytes = rom.subarray(funcStart, funcStart + 200);
console.log('Function bytes at 0x03FC1C:');
for (let i = 0; i < bytes.length; i += 16) {
  const hex = Array.from(bytes.subarray(i, Math.min(i+16, bytes.length)))
    .map(b => b.toString(16).padStart(2,'0')).join(' ');
  console.log(`  ${(funcStart + i).toString(16).padStart(6,'0')}: ${hex}`);
}

// Dump the 64-byte table
const table = rom.subarray(0x03FC41, 0x03FC81);
console.log('\n8x8 scan-to-key table at 0x03FC41:');
for (let row = 0; row < 8; row++) {
  const rowHex = Array.from(table.subarray(row*8, row*8+8))
    .map(b => b.toString(16).padStart(2,'0')).join(' ');
  console.log(`  row ${row}: ${rowHex}`);
}

// Search for callers: CALL 0x03FC1C = CD 1C FC 03
for (let i = 0; i < rom.length - 3; i++) {
  if (rom[i] === 0xCD && rom[i+1] === 0x1C && rom[i+2] === 0xFC && rom[i+3] === 0x03) {
    console.log(`CALL 0x03FC1C at 0x${i.toString(16).padStart(6,'0')}`);
  }
}

// Search for LD A,(D0058D) = 3A 8D 05 D0
for (let i = 0; i < rom.length - 3; i++) {
  if (rom[i] === 0x3A && rom[i+1] === 0x8D && rom[i+2] === 0x05 && rom[i+3] === 0xD0) {
    console.log(`LD A,(D0058D) at 0x${i.toString(16).padStart(6,'0')}`);
  }
}
```
